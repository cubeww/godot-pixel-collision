cmake_minimum_required(VERSION 3.21)
project(godot_pixel_collision VERSION 0.1.0 LANGUAGES CXX)
file(WRITE "${CMAKE_BINARY_DIR}/.gdignore" "")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "MSVC runtime" FORCE)
endif()
option(PIXEL_COLLISION_BUILD_EXTENSION "Build the Godot GDExtension" ON)
option(PIXEL_COLLISION_BUILD_TESTS "Build the standalone core tests" ON)

file(GLOB PIXEL_CORE_SOURCES CONFIGURE_DEPENDS "src/core/*.cpp")
add_library(pixel_collision_core STATIC ${PIXEL_CORE_SOURCES})
target_include_directories(pixel_collision_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
set_target_properties(pixel_collision_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(pixel_collision_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(pixel_collision_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(PIXEL_COLLISION_BUILD_TESTS)
    enable_testing()
    add_executable(pixel_collision_tests tests/core_tests.cpp)
    target_link_libraries(pixel_collision_tests PRIVATE pixel_collision_core)
    add_test(NAME pixel_collision_core COMMAND pixel_collision_tests)
endif()

if(PIXEL_COLLISION_BUILD_EXTENSION)
    set(GODOT_CPP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/godot-cpp" CACHE PATH "Path to godot-cpp")
    if(NOT EXISTS "${GODOT_CPP_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "Missing godot-cpp. Run: git submodule update --init --recursive")
    endif()
    set(GODOTCPP_TARGET "template_release" CACHE STRING "Godot binding target" FORCE)
    set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "Core validation uses caught exceptions" FORCE)
    set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "Binding class profile" FORCE)
    add_subdirectory("${GODOT_CPP_DIR}" "godot-cpp")
    file(GLOB PIXEL_GODOT_SOURCES CONFIGURE_DEPENDS "src/godot/*.cpp")
    add_library(pixel_collision SHARED ${PIXEL_GODOT_SOURCES})
    target_link_libraries(pixel_collision PRIVATE pixel_collision_core godot::cpp)
    target_include_directories(pixel_collision PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    if(MSVC)
        target_compile_options(pixel_collision PRIVATE /W4 /utf-8)
    else()
        target_compile_options(pixel_collision PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(WIN32)
        set(PIXEL_PLATFORM "windows")
    elseif(APPLE)
        set(PIXEL_PLATFORM "macos")
    else()
        set(PIXEL_PLATFORM "linux")
    endif()
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" PIXEL_PROCESSOR)
    if(PIXEL_PROCESSOR MATCHES "^(aarch64|arm64)$")
        set(PIXEL_ARCH "arm64")
    else()
        set(PIXEL_ARCH "x86_64")
    endif()
    set_target_properties(pixel_collision PROPERTIES PREFIX ""
        OUTPUT_NAME "pixel_collision.${PIXEL_PLATFORM}.${PIXEL_ARCH}"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/addons/pixel_collision/bin"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/addons/pixel_collision/bin"
        RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/addons/pixel_collision/bin"
        LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/addons/pixel_collision/bin")
endif()
